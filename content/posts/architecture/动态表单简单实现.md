动态表单简单版 后台维护 一个类就是一条数据就是一个字段  
然后通过接口给到前端 前端进行渲染 字段必填 类型都可以自定义  
数据存储于mongo 方便扩展字段  

```
@Data
@EqualsAndHashCode(callSuper = true)
@JsonInclude(JsonInclude.Include.NON_NULL)
public class Head {
    //表单名称
    private String formName;
    //序号
    private Integer serialNumber;
    //唯一键
    private String uniqueKey;
    //字段名
    private String fieldName;
    //变量名("枚举的时候需要")，对应数据字典的type
    private String variableCode="";
    //变量名对应中文
    private String variableName;
    //类型 1.输入框   2.枚举  3.请求接口  4.时间戳  5.text输入框
    private Integer type=1;
    //长度
    private Integer length=60;

    //新增、修改页面时是否展示 0不展示  1展示
    private Integer show=1;

    //是否可修改删除  0不可以   1可以
    private Integer canRemove=1;

    //是否在表头列表页面展示  0不可以   1可以
    private Integer showOnEditPage=1;
}
```
