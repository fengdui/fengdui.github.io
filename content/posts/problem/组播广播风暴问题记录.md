网管让我们配置一下 不然公司网络太卡了
# 添加规则，丢弃从ens33进入的组播流量
sudo iptables -I INPUT -i ens33 -d 224.0.0.0/4 -j DROP
# 添加规则，丢弃从ens33转发的组播流量
sudo iptables -I FORWARD -i ens33 -d 224.0.0.0/4 -j DROP
# 添加规则，丢弃从ens33输出的组播流量（如果需要）
sudo iptables -I OUTPUT -o ens33 -d 224.0.0.0/4 -j DROP

# 4. 特别过滤SSDP地址（239.255.255.250）和mDNS（224.0.0.251）
sudo iptables -I INPUT -d 239.255.255.250 -j DROP  
sudo iptables -I FORWARD -d 239.255.255.250 -j DROP  
sudo iptables -I OUTPUT -d 239.255.255.250 -j DROP

sudo iptables -I INPUT -d 224.0.0.251 -j DROP  
sudo iptables -I FORWARD -d 224.0.0.251 -j DROP  
sudo iptables -I OUTPUT -d 224.0.0.251 -j DROP

# 5. 停止组播相关服务（如avahi-daemon）
echo "停止组播服务..."  
sudo systemctl stop avahi-daemon 2>/dev/null

echo "配置完成！"  
echo "当前iptables规则："  
sudo iptables -L INPUT -n | grep DROP  
sudo iptables -L FORWARD -n | grep DROP  
sudo iptables -L OUTPUT -n | grep DROP
